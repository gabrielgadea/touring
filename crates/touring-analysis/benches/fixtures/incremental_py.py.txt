# Stable Python fixture for incremental update benchmarks (~1 KB).

from __future__ import annotations
import re
import hashlib
from typing import Optional, Any


@dataclass
class Config:
    name: str
    max_retries: int = 3
    timeout: float = 5.0
    enabled: bool = True

    def validate(self) -> list[str]:
        errors = []
        if not self.name:
            errors.append("name is required")
        if self.max_retries < 1:
            errors.append("max_retries must be >= 1")
        return errors


API_VERSION = "2.1.0"


def is_valid_email(email: str) -> bool:
    pattern = r"^[^\s@]+@[^\s@]+\.[^\s@]+$"
    return bool(re.match(pattern, email.strip()))


def sanitize(text: str, max_len: int = 200) -> str:
    return re.sub(r"[<>]", "", text).strip()[:max_len]


class Cache:
    def __init__(self, capacity: int = 100) -> None:
        self._data: dict[str, Any] = {}
        self._capacity = capacity
        self._access_order: list[str] = []

    def get(self, key: str) -> Optional[Any]:
        if key in self._data:
            self._access_order.remove(key)
            self._access_order.append(key)
            return self._data[key]
        return None

    def put(self, key: str, value: Any) -> None:
        if key in self._data:
            self._access_order.remove(key)
        elif len(self._data) >= self._capacity:
            oldest = self._access_order.pop(0)
            del self._data[oldest]
        self._data[key] = value
        self._access_order.append(key)

    def size(self) -> int:
        return len(self._data)


def compute_hash(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def merge(base: dict[str, Any], update: dict[str, Any]) -> dict[str, Any]:
    result = base.copy()
    for k, v in update.items():
        if isinstance(v, dict) and k in result and isinstance(result[k], dict):
            result[k] = merge(result[k], v)
        else:
            result[k] = v
    return result
